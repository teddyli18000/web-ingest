cmake_minimum_required(VERSION 3.18)
project(aes_xag_probe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(MOCKTURTLE_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(MOCKTURTLE_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(MOCKTURTLE_BUILD_EXPERIMENTS OFF CACHE BOOL "" FORCE)
add_subdirectory("${MOCKTURTLE_DIR}" "${CMAKE_BINARY_DIR}/mockturtle" EXCLUDE_FROM_ALL)
add_executable(aes_xag_probe mockturtle_aes_probe.cpp)
target_link_libraries(aes_xag_probe PRIVATE mockturtle)
